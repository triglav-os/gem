cmake_minimum_required(VERSION 3.20)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(gem_samples C)
    set(CMAKE_C_STANDARD 11)
    set(CMAKE_C_STANDARD_REQUIRED ON)
    set(CMAKE_C_EXTENSIONS OFF)
    if(NOT CMAKE_BUILD_TYPE)
        set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
    endif()
    set(GEM_SDK_ROOT "" CACHE PATH "GEM SDK containing include/ and lib/")
    include("${GEM_SDK_ROOT}/gem_sdk_config.cmake" OPTIONAL)
    find_path(GEM_SDK_INCLUDE_DIR gem.h HINTS ${GEM_SDK_ROOT}/include REQUIRED)
    foreach(name platform_linux vdi aes gem)
        find_library(GEM_${name}_LIBRARY NAMES ${name}
            HINTS ${GEM_SDK_ROOT}/lib REQUIRED)
        add_library(${name} SHARED IMPORTED)
        set_target_properties(${name} PROPERTIES
            IMPORTED_LOCATION "${GEM_${name}_LIBRARY}"
            INTERFACE_INCLUDE_DIRECTORIES "${GEM_SDK_INCLUDE_DIR}")
    endforeach()
    set_target_properties(gem PROPERTIES INTERFACE_LINK_LIBRARIES platform_linux)
    set_target_properties(aes PROPERTIES INTERFACE_LINK_LIBRARIES vdi)
    if(GEM_SDK_PLATFORM STREQUAL "rasta")
        find_library(GEM_rasta_LIBRARY NAMES rasta HINTS ${GEM_SDK_ROOT}/lib REQUIRED)
        add_library(rasta SHARED IMPORTED)
        set_target_properties(rasta PROPERTIES IMPORTED_LOCATION "${GEM_rasta_LIBRARY}")
        set_target_properties(vdi PROPERTIES INTERFACE_LINK_LIBRARIES "rasta;m")
    else()
        set_target_properties(vdi PROPERTIES INTERFACE_LINK_LIBRARIES "platform_linux;m")
    endif()
    get_filename_component(GEM_LIBRARY_DIR "${GEM_gem_LIBRARY}" DIRECTORY)
    set(GEM_SAMPLES_DIR "${CMAKE_BINARY_DIR}/runtime" CACHE PATH "Sample executable output")
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
endif()

set(SAMPLES_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${GEM_SAMPLES_DIR}/lib)
set(CMAKE_BUILD_WITH_INSTALL_RPATH FALSE)
set(CMAKE_BUILD_RPATH "${GEM_LIBRARY_DIR};${GEM_SAMPLES_DIR}/lib")
set(CMAKE_INSTALL_RPATH "$ORIGIN/lib;$ORIGIN/../lib")
include_directories(${SAMPLES_ROOT}/include)
add_subdirectory(data)
add_subdirectory(lib)
add_subdirectory(src)
